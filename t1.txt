// BROWSER FINGERPRINTING (Anti-Incognito Device ID)
// =========================================
function generateDeviceFingerprint() {
    const nav = window.navigator;
    const scr = window.screen;
    
    // 1. Collect stable browser and hardware properties
    const components = [
        nav.userAgent || 'unknown',
        nav.language || 'unknown',
        scr.width || 0,
        scr.height || 0,
        scr.colorDepth || 0,
        new Date().getTimezoneOffset() || 0,
        nav.hardwareConcurrency || 0, // Number of CPU cores
        nav.platform || 'unknown'
    ];
    
    // 2. Combine into a single raw string
    const rawString = components.join('|');
    
    // 3. Fast hash function (djb2) to convert the string into a unique hex ID
    let hash = 5381;
    for (let i = 0; i < rawString.length; i++) {
        hash = ((hash << 5) + hash) + rawString.charCodeAt(i);
        hash = hash & hash; // Convert to 32bit integer
    }
    
    // Return formatted ID (e.g., "fp_a1b2c3d4")
    return 'fp_' + (hash >>> 0).toString(16).padStart(8, '0');
}

function getDeviceId() {
    // Try to get existing ID from localStorage
    let deviceId = localStorage.getItem('aries_device_id');
    
    if (!deviceId) {
        // If not found (or in Incognito), generate a new one based on fingerprint
        deviceId = generateDeviceFingerprint();
        
        // Save it so it doesn't recalculate every time
        localStorage.setItem('aries_device_id', deviceId);
    }
    
    return deviceId;
}

async function postToBackend(payload) {
  let idToken = "";
  if (auth.currentUser) {
    try { idToken = await auth.currentUser.getIdToken(); } catch (e) {}
  }
  payload.idToken = idToken;

  let appCheckToken = "";
  try {
    const appCheckTokenResult = await appCheck.getToken();
    appCheckToken = appCheckTokenResult.token;
  } catch (e) {
    console.error("App Check getToken failed:", e);
  }

  if (!appCheckToken) {
    throw new Error("App Check token unavailable. Check the reCAPTCHA site key and that this domain is registered in Firebase App Check.");
  }

  const response = await fetch(BACKEND_URL, {
    method: "POST",
    headers: {
      "Content-Type": "application/json",
      "X-Firebase-AppCheck": appCheckToken
    },
    body: JSON.stringify(payload)
  });

  const text = await response.text();
  let data;
  try {
    data = JSON.parse(text);
  } catch (e) {
    throw new Error("Server returned non-JSON (HTTP " + response.status + "): " + text.slice(0, 200));
  }

  if (data && typeof data === "object") {
    Object.defineProperty(data, "__ok", { value: response.ok, enumerable: false });
    Object.defineProperty(data, "__status", { value: response.status, enumerable: false });
  }
  return data;
}

async function handleLogin(email, password, status) {
  if (!email) { showErrorModal("Please enter your email address."); return; }
  if (!password) { showErrorModal("Please enter your password."); return; }

  status.innerHTML = '<span class="spinner"></span>Logging in...';

  try {
    // 1. Password first. Session exists but carries no MFA claim yet.
    const cred = await auth.signInWithEmailAndPassword(email, password);

    if (!cred.user.emailVerified) {
      await auth.signOut();
      showErrorModal("Please verify your email before logging in.");
      status.innerText = "";
      return;
    }

    const checkData = await postToBackend({ action: "check2FARequired" });

    if (checkData.requires2FA) {
      document.getElementById("twoFALoginCode").value = "";
      document.getElementById("twoFALoginModal").style.display = "flex";
      status.innerText = "";
      return; // session is unprivileged until complete2FALogin runs
    }

    status.innerText = "Logged in";
    document.getElementById("authLoginForm").reset();

  } catch (error) {
    console.error(error);
    let message = "Login failed";
    if (error.code === "auth/user-not-found" || error.code === "auth/wrong-password" || error.code === "auth/invalid-credential") {
      message = "Invalid email or password";
    } else if (error.code === "auth/too-many-requests") {
      message = "Too many failed attempts. Try again later";
    }
    showErrorModal(message);
    status.innerText = "";
  }
}

// Load saved conversation on page load
async function loadSavedConversation() {
  try {
    const data = await postToBackend({
      action: "loadAIConversation",
      deviceId: deviceId
    });
	
    if (data.success && data.messages && data.messages.length > 0) {
      // Restore conversation (system prompt is always re-added)
      aiConversation = [SYSTEM_PROMPT, ...data.messages];
      
      // Re-render chat UI
      data.messages.forEach(msg => {
        if (msg.role === "user" || msg.role === "assistant") {
          appendChatMessage(msg.role, msg.content);
        }
      });
    }
  } catch (err) {
    console.error("Failed to load conversation:", err);
  }
}

function updateRemainingDisplay(remaining) {
  const element = document.getElementById("aiRemaining");
  if (!element) return;
  
  element.classList.remove("low", "empty");
  
  if (remaining <= 0) {
    element.innerText = "0/10 — LIMIT REACHED";
    element.classList.add("empty");
  } else if (remaining <= 3) {
    element.innerText = `${remaining}/10 messages left today`;
    element.classList.add("low");
  } else {
    element.innerText = `${remaining}/10 messages left today`;
  }
}

async function loadRemainingCount() {
  try {
    const data = await postToBackend({
      action: "getRemainingAI",
      deviceId: deviceId
    });
	
    if (data.success && typeof data.remaining === "number") {
      updateRemainingDisplay(data.remaining);
    } else {
      updateRemainingDisplay(10);
    }
  } catch (err) {
    console.error("Failed to load remaining count:", err);
    updateRemainingDisplay(10);
  }
}

async function askAriesAi(event) {
  event.preventDefault();
  const promptInput = document.getElementById("aiPrompt");
  const status = document.getElementById("aiStatus");
  const button = document.getElementById("aiButton");
  const prompt = promptInput.value.trim();

  if (!prompt) { status.innerText = "Enter a prompt"; return; }

  appendChatMessage("user", prompt);
  const aiMessage = appendChatMessage("ai", "Thinking...");
  aiConversation.push({ role: "user", content: prompt });
  promptInput.value = "";
  status.innerText = "Thinking...";
  button.disabled = true;

  try {
	const data = await postToBackend({
      action: "askAI",
      messages: aiConversation,
      temperature: 0.7,
      max_tokens: 1024,
      deviceId: deviceId
    });

    if (typeof data.remaining === "number") {
      updateRemainingDisplay(data.remaining);
    }

    if (!data.__ok) {
      throw new Error(data.message || "AI returned HTTP " + data.__status);
    }

    const reply = data.choices && data.choices[0] && data.choices[0].message ? data.choices[0].message.content : null;
    
    if (!reply) {
      console.error("Unexpected response:", data);
      throw new Error("AI returned empty response");
    }
    
    aiMessage.querySelector(".chat-bubble").innerText = reply;
    aiConversation.push({ role: "assistant", content: reply });
    scrollAiResponse();
    status.innerText = "Complete";
	// Save conversation to backend (fire-and-forget, don't block UI)
    saveConversationToBackend();
  } catch (err) {
    console.error("AI Error:", err);
    aiConversation.pop();
    aiMessage.querySelector(".chat-bubble").innerText = "Unable to connect to Aries AI: " + err.message;
    scrollAiResponse();
    status.innerText = "AI unavailable";
  } finally {
    button.disabled = false;
  }
}

loadRemainingCount();

async function updateWaitlistCount() {
  const counterElement = document.getElementById("waitlist-counter");
  try {
    const statsDoc = await firebase.firestore().collection("metadata").doc("stats").get();
    if (statsDoc.exists && statsDoc.data().waitlistCount > 0) {
      counterElement.innerText = statsDoc.data().waitlistCount + " people have joined the waitlist!";
    } else {
      counterElement.innerText = "Be the first to join the waitlist!";
    }
  } catch (error) {
    console.error("Error fetching count (using fallback):", error);
    counterElement.innerText = "Join the exclusive waitlist today!";
  }
}

function appendChatMessage(role, text) {
  const responseWindow = document.getElementById("aiResponse");
  const message = document.createElement("div");
  const bubble = document.createElement("div");
  message.className = "chat-message " + role;
  bubble.className = "chat-bubble";
  bubble.innerText = text;
  message.appendChild(bubble);
  responseWindow.appendChild(message);
  scrollAiResponse();
  return message;
}

function scrollAiResponse() {
  const responseWindow = document.getElementById("aiResponse");
  responseWindow.scrollTop = responseWindow.scrollHeight;
}

function toggleAiExpand() {
  const aiConsole = document.querySelector(".ai-console");
  const button = document.getElementById("aiExpandButton");
  const accountToggle = document.querySelector(".account-toggle");
  const isExpanded = aiConsole.classList.toggle("ai-expanded");
  document.body.classList.toggle("no-scroll", isExpanded);
  
  // Temporarily hide the Account button when expanded to prevent overlap
  if (accountToggle) {
    accountToggle.style.display = isExpanded ? "none" : "block";
  }

  if (!isExpanded) {
    aiConsole.style.width = ""; 
    aiConsole.style.height = ""; 
    aiConsole.style.left = ""; 
    aiConsole.style.top = ""; 
    aiConsole.style.right = ""; 
    aiConsole.style.bottom = "";
  }
  
  button.innerText = isExpanded ? "Normal" : "Expand";
  scrollAiResponse();
}
